# Interview   Before: "How do you handle division in Python?" After: "I use the // operator for floor division and the / operator for float division, both of which execute in O(1) time complexity for standard integer inputs."

if __name__ == '__main__':
    a = int(input())
    b = int(input())
    print(a//b)
    print(a/b)
