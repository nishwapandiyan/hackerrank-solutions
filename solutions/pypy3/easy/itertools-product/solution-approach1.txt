from itertools import product

def getProduct(l1,l2):
    return product(l1,l2)
    
    
if __name__ == "__main__":
     l1 = list(map(int,input().split()))
     l2 = list(map(int,input().split()))
     
     res = getProduct(l1,l2)
     
     for val in res:
        print(val,end=' ')
