n,m = map(int,input().split())

for i in range(n//2):
    pat = ".|." * (2*i+1)
    print(pat.center(m,'-'))
    
print("WELCOME".center(m,'-'))
    
for i in range(n//2-1,-1,-1):
    pat = ".|." * (2*i+1)
    print(pat.center(m,'-'))
