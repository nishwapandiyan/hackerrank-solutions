# Interview   Before: "How would you replace spaces with hyphens in a string?" After: "I used split() and join() to achieve this in O(n) time, where n is the string length, ensuring all whitespace sequences are handled correctly."
# Pitfalls    (1) Using split(' ') instead of split() would incorrectly handle multiple consecutive spaces by creating empty strings in the resulting list.  (2) Assuming the input string contains no leading or trailing whitespace when using split(' ') instead of the default split() method.
# ──────────────────────────────────────────────────



def split_and_join(line):
    line = line.split()
    line = "-".join(line)
    return line

if __name__ == '__main__':
    line = input()
    result = split_and_join(line)
    print(result)
