if __name__ == '__main__':
    Students = []
    for _ in range(int(input())):
        name = input()
        score = float(input())
        Students.append([name,score])
    grades = sorted(set([g for name,g in Students]))
    second_lowest = grades[1]
        
    ne = [name for name,g in Students if g == second_lowest]
        
    for i in sorted(ne):
        print(i)
