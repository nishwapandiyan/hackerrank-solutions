from collections import Counter

shoes = int(input())
shoe_sizes = list(map(int,input().split(" ")))

shoc_sizes = Counter(shoe_sizes)

customers = int(input())
total = 0

for i in range(customers):
    size,price = map(int,input().split(" "))
    
    if(shoc_sizes[size] > 0):
      total += price
      shoc_sizes[size] -= 1
    
print(total)
